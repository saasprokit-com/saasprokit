# Technical SEO Checklist

## Metadata API

### Static Metadata

| Status | Task | File |
|--------|------|------|
| [ ] | Add root metadata | `app/layout.tsx` |
| [ ] | Title with template | `title: { default, template }` |
| [ ] | Description (150-160 chars) | `description` |
| [ ] | Open Graph tags | `openGraph` |
| [ ] | Twitter Card tags | `twitter` |
| [ ] | Robots configuration | `robots` |
| [ ] | Google verification | `verification.google` |

```typescript
// app/layout.tsx
import type { Metadata } from 'next'

export const metadata: Metadata = {
  title: {
    default: 'Your Site Name',
    template: '%s | Your Site Name'
  },
  description: 'Compelling description (150-160 characters)',
  openGraph: {
    type: 'website',
    locale: 'en_US',
    url: 'https://yourdomain.com',
    siteName: 'Your Site Name',
    images: [{
      url: 'https://yourdomain.com/og-image.jpg',
      width: 1200,
      height: 630,
      alt: 'OG Image Alt Text',
    }],
  },
  twitter: {
    card: 'summary_large_image',
    site: '@yourusername',
  },
  robots: {
    index: true,
    follow: true,
    googleBot: {
      index: true,
      follow: true,
      'max-video-preview': -1,
      'max-image-preview': 'large',
      'max-snippet': -1,
    },
  },
  verification: {
    google: 'your-verification-code',
  },
}
```

### Dynamic Metadata

| Status | Task | Route |
|--------|------|-------|
| [ ] | Blog posts | `app/blog/[slug]/page.tsx` |
| [ ] | Product pages | `app/products/[slug]/page.tsx` |
| [ ] | Author pages | `app/authors/[slug]/page.tsx` |
| [ ] | Category pages | `app/category/[slug]/page.tsx` |

```typescript
// app/blog/[slug]/page.tsx
import type { Metadata } from 'next'

export async function generateMetadata({ params }): Promise<Metadata> {
  const post = await fetchPost(params.slug)

  return {
    title: post.title,
    description: post.excerpt,
    openGraph: {
      title: post.title,
      description: post.excerpt,
      type: 'article',
      publishedTime: post.publishedAt,
      authors: [post.author],
      images: [{
        url: post.coverImage,
        width: 1200,
        height: 630,
        alt: post.title,
      }],
    },
    alternates: {
      canonical: `https://yourdomain.com/blog/${params.slug}`,
    },
  }
}
```

## Sitemap

| Status | Task | File |
|--------|------|------|
| [ ] | Create dynamic sitemap | `app/sitemap.ts` |
| [ ] | Include all static pages | Priority 1.0 |
| [ ] | Include blog posts | Priority 0.7 |
| [ ] | Include products | Priority 0.9 |
| [ ] | Set correct `lastModified` | Accuracy |

```typescript
// app/sitemap.ts
import { MetadataRoute } from 'next'

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const baseUrl = 'https://yourdomain.com'
  const posts = await getAllPosts()

  const staticPages = ['', '/about', '/contact'].map((route) => ({
    url: `${baseUrl}${route}`,
    lastModified: new Date(),
    changeFrequency: 'monthly' as const,
    priority: route === '' ? 1 : 0.8,
  }))

  const blogPages = posts.map((post) => ({
    url: `${baseUrl}/blog/${post.slug}`,
    lastModified: new Date(post.updatedAt),
    changeFrequency: 'weekly' as const,
    priority: 0.7,
  }))

  return [...staticPages, ...blogPages]
}
```

## robots.txt

| Status | Task | File |
|--------|------|------|
| [ ] | Create robots.ts | `app/robots.ts` |
| [ ] | Allow public pages | `/` |
| [ ] | Disallow admin/api | `/api/`, `/admin/` |
| [ ] | Link to sitemap | `sitemap` property |

```typescript
// app/robots.ts
import { MetadataRoute } from 'next'

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        userAgent: '*',
        allow: '/',
        disallow: ['/api/', '/admin/', '/private/'],
      },
    ],
    sitemap: 'https://yourdomain.com/sitemap.xml',
  }
}
```
